⚙️ Parámetros de Conexión

📋 Apps Script para Google Sheets

Pega esto en Extensiones → Apps Script
// BiG Inventarios · Apps Script
// Pega este código en Extensiones → Apps Script de tu Google Sheet
// Luego: Implementar → Nueva implementación → Aplicación web
// Ejecutar como: Yo | Quién tiene acceso: Cualquier usuario

const SHEET_ID = '1PsEEwR16EN1VnRqh6eyif87295QN4g9tAOP4-QfiFoU';

function doGet(e) {
  const sheet = e.parameter.sheet || 'equipos_informatica';
  const ss = SpreadsheetApp.openById(SHEET_ID);
  const ws = ss.getSheetByName(sheet);
  if (!ws) return json({error: 'Hoja no encontrada: ' + sheet});
  const rows = ws.getDataRange().getValues();
  const headers = rows[0];
  const data = rows.slice(1).map(row => {
    const obj = {};
    headers.forEach((h, i) => obj[h] = row[i]);
    return obj;
  });
  return json({ok: true, data, sheet, total: data.length});
}

function doPost(e) {
  const payload = JSON.parse(e.postData.contents);
  const {sheet, action, row, rowIndex} = payload;
  const ss = SpreadsheetApp.openById(SHEET_ID);
  const ws = ss.getSheetByName(sheet);
  if (!ws) return json({error: 'Hoja no encontrada'});
  
  if (action === 'append') {
    const headers = ws.getRange(1, 1, 1, ws.getLastColumn()).getValues()[0];
    const newRow = headers.map(h => row[h] !== undefined ? row[h] : '');
    ws.appendRow(newRow);
    return json({ok: true, action: 'appended'});
  }
  
  if (action === 'update' && rowIndex >= 0) {
    const headers = ws.getRange(1, 1, 1, ws.getLastColumn()).getValues()[0];
    const updRow = headers.map(h => row[h] !== undefined ? row[h] : '');
    ws.getRange(rowIndex + 2, 1, 1, updRow.length).setValues([updRow]);
    return json({ok: true, action: 'updated'});
  }
  
  return json({error: 'Acción no reconocida'});
}

function json(data) {
  return ContentService
    .createTextOutput(JSON.stringify(data))
    .setMimeType(ContentService.MimeType.JSON);
}
